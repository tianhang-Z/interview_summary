#### future/promise

原理

1. **创建 `promise` 和 `future`**：通过 `promise.get_future()` 获取关联的 `future`。
2. **生产者线程设置结果**：生产者通过 `promise.set_value()` 存储结果（或 `set_exception()` 存储异常），此时共享状态变为 “就绪”。
3. **消费者线程获取结果**：消费者通过 `future.get()` 获取结果（若未就绪则阻塞等待），或通过 `future.wait()` 主动等待结果就绪后再处理。

```c++
#include <iostream>
#include <future>
#include <thread>
#include <chrono>

// 生产者线程：计算结果并通过promise传递
// 内部完美转发
void produce_result(std::promise<int>& prom) {
    std::this_thread::sleep_for(std::chrono::seconds(2)); // 模拟耗时计算
    int result = 42; // 生成结果
    prom.set_value(result); // 将结果存入promise（共享状态变为就绪）
}

int main() {
    // 1. 创建promise和关联的future
    std::promise<int> prom;
    std::future<int> fut = prom.get_future(); // future与promise绑定

    // 2. 启动生产者线程，传递promise（注意：需移动语义，因为promise不可复制）
    std::thread producer(produce_result, std::move(prom));

    // 3. 消费者线程（主线程）获取结果
    std::cout << "等待结果中..." << std::endl;
    int value = fut.get(); // 若结果未就绪，此处会阻塞等待
    std::cout << "获取到结果：" << value << std::endl;

    producer.join();
    return 0;
}
```

## 手撕线程池

```c++
/*
线程池 含有一组线程和一个任务队列    线程会从队列自动取任务执行
队列未满时 可以向队列中添加任务 runTask()
队列有任务时 可以从队列中取出任务 take()
*/
class ThreadPool : noncopyable {
 public:
  explicit ThreadPool(size_t threadNum, size_t maxQueueSize = 65536,
                      const ThreadInitCallback& callback = nullptr);
  ~ThreadPool();

  void runTask(const Task&);
  void runTask(Task&&);
  void stop();
  size_t threadNum() const;

 private:
  //每个线程执行的函数 其会从任务队列中取出任务
  void runInThread(size_t index);
  Task take();

  using ThreadPtr = std::unique_ptr<std::thread>;
  using ThreadList = std::vector<ThreadPtr>;

  ThreadList threads_;
  std::mutex mutex_;
  std::condition_variable notEmpty_;
  std::condition_variable notFull_;
  std::queue<Task> taskQueue_;
  const size_t maxQueueSize_;
  std::atomic_bool running_;
  ThreadInitCallback threadInitCallback_;
};

ThreadPool::ThreadPool(size_t threadNum, size_t maxQueueSize,
                       const ThreadInitCallback& callback)
    : maxQueueSize_(maxQueueSize),
      running_(true),
      threadInitCallback_(callback) {
  assert(maxQueueSize_ > 0);
  for (size_t i = 0; i < threadNum; ++i) {
    // 从0开始标号，作为线程池中线程的标识
    threads_.emplace_back(new std::thread([i, this]() { runInThread(i); }));
  }
  TRACE("ThreadPool construct threadNum {}, maxQueueSize {}", threadNum,
        maxQueueSize);
}

ThreadPool::~ThreadPool() {
  if (running_) {
    stop();
  }
  TRACE("ThreadPool destruct");
}

/*
任务提交到队列
*/
void ThreadPool::runTask(const Task& task) {
  assert(running_);
  if (threads_.empty()) {
    // 如果没有线程池线程，则直接在当前线程中执行任务。
    task();
  } else {
    std::unique_lock<std::mutex> lock(mutex_);
    while (taskQueue_.size() >= maxQueueSize_) {
      notFull_.wait(lock);
    }
    taskQueue_.push(task);
    notEmpty_.notify_one();
  }
}
void ThreadPool::runTask(Task&& task) {
  assert(running_);
  if (threads_.empty()) {
    task();
  } else {
    std::unique_lock<std::mutex> lock(mutex_);
    while (taskQueue_.size() >= maxQueueSize_) {
      notFull_.wait(lock);
    }
    taskQueue_.push(std::move(task));
    notEmpty_.notify_one();
  }
}

void ThreadPool::stop() {
  assert(running_);
  running_ = false;
  {
    std::lock_guard<std::mutex> lock(mutex_);
    notEmpty_.notify_all();
  }
  for (auto& thread : threads_) {
    thread->join();
  }
}

size_t ThreadPool::threadNum() const { return threads_.size(); }

/*
每个线程执行的函数 其会从任务队列中取出任务 index作为线程池中线程的标识
*/
void ThreadPool::runInThread(size_t index) {
  if (threadInitCallback_) {
    threadInitCallback_(index);
  }
  while (running_) {
    if (Task task = take()) {
      task();
    }
  }
}

// 线程池中线程执行的函数 取任务
Task ThreadPool::take() {
  std::unique_lock<std::mutex> lock(mutex_);
  // 确保能在停止指令发出时，正确停止，返回一个空Task
  while (taskQueue_.empty() && running_) {
    notEmpty_.wait(lock);
  }
  Task task;
  if (!taskQueue_.empty()) {
    task = taskQueue_.front();
    taskQueue_.pop();
    notFull_.notify_one();
  }
  return task;
}
```

## 如何设计智能指针 

### unique_ptr

**删除拷贝构造和拷贝赋值**，禁止了对 `unique_ptr` 对象的拷贝行为，从而避免多个 `unique_ptr` 同时持有同一资源

### shared_ptr

`shared_ptr` 的设计目标是管理**动态分配（堆上）的资源**，其析构函数会自动调用 `delete` 释放所管理的指针。

而**栈上的对象**会在其作用域结束时**自动销毁**（由编译器管理），无需也不能手动调用 `delete`。

* 同一裸指针不能被多个 `shared_ptr` 独立接管，否则会导致重复释放：

  该错误不会在编译期检测出来

  ```
  int* raw_ptr = new int(100);
  std::shared_ptr<int> sp1(raw_ptr);
  std::shared_ptr<int> sp2(raw_ptr);  // 错误！sp1 和 sp2 会各自认为自己是唯一管理者，析构时重复 delete
  ```

* 正确做法是通过拷贝 `shared_ptr` 来共享所有权：

  ```
  std::shared_ptr<int> sp1(raw_ptr);
  std::shared_ptr<int> sp2 = sp1;  // 正确！通过拷贝共享，引用计数变为 2
  ```

##### 初始化方法

* 用裸指针初始化（不推荐，需谨慎）

  ```
      int* raw_ptr = new int(100);  // 裸指针（堆上对象）
      std::shared_ptr<int> sp(raw_ptr);  // 初始化：接管 raw_ptr 的所有权
  ```

* 通过 `std::make_shared` 初始化

  ```
      auto sp1 = std::make_shared<int>(100);
  ```

* 通过拷贝构造或赋值初始化（共享所有权）

  ```
      auto sp1 = std::make_shared<int>(100);  // 计数 = 1
  
      // 拷贝构造：sp2 与 sp1 共享资源，计数 = 2
      std::shared_ptr<int> sp2(sp1);
  
      // 拷贝赋值：sp3 与 sp1 共享资源，计数 = 3
      std::shared_ptr<int> sp3;
      sp3 = sp1;
  ```

* 通过移动构造或赋值初始化（转移临时所有权）

  ```
      auto sp1 = std::make_shared<int>(100);  // 计数 = 1
      std::cout << "sp1 是否为空: " << (sp1 == nullptr) << std::endl;  // 输出 0（非空）
  
      // 移动构造：sp2 接管 sp1 的资源，sp1 变为空
      std::shared_ptr<int> sp2(std::move(sp1));
      std::cout << "sp1 是否为空: " << (sp1 == nullptr) << std::endl;  // 输出 1（空）
      std::cout << "sp2 计数: " << sp2.use_count() << std::endl;  // 输出 1
  
      // 移动赋值：sp3 接管 sp2 的资源，sp2 变为空
      std::shared_ptr<int> sp3;
      sp3 = std::move(sp2);
  ```

  

##### 设计方法：

基于控制块，其中负责引用计数

```
+----------------+       +-------------------+
|  shared_ptr A  |       |                   |
| -------------- |       |   控制块          |
| ptr ---------->+------>+   - use_count: 3  |
| control_block->+------>+   - weak_count: 2 |
+----------------+       |   - deleter: ...  |
                         |                   |
+----------------+       +-------------------+
|  shared_ptr B  |               ^
| -------------- |               |
| ptr ---------->+---------------+
| control_block->+---------------+
+----------------+
                                  
+----------------+       +-------------------+
|   weak_ptr C   |       |   资源 (int 10)   |
| -------------- |       |                   |
| control_block->+------>+                   |
+----------------+       +-------------------+
```

```c
#include <cstddef>
#include <utility>
#include <atomic>
#include <functional>  // 用于 std::function（存储删除器）

// 前置声明
template <typename T>
class WeakPtr;

// 控制块基类（用于多态管理不同类型的删除器）
class ControlBlockBase {
public:
    std::atomic<size_t> use_count;    // 强引用计数（shared_ptr 数量）
    std::atomic<size_t> weak_count;   // 弱引用计数（weak_ptr 数量）

    ControlBlockBase() : use_count(1), weak_count(1) {}  // 初始：1个shared_ptr，控制块自身算1个weak_count
    virtual ~ControlBlockBase() = default;  // 虚析构，确保派生类正确释放
    virtual void destroy() = 0;  // 纯虚函数：释放资源（由派生类实现）
};

// 控制块派生类（针对特定类型 T 和删除器）
template <typename T, typename Deleter>
class ControlBlock : public ControlBlockBase {
private:
    T* ptr;               // 指向资源的指针
    Deleter deleter;      // 自定义删除器

public:
    // 构造：接收资源指针和删除器
    ControlBlock(T* p, Deleter d) : ptr(p), deleter(std::move(d)) {}

    // 释放资源（调用删除器）
    void destroy() override {
        deleter(ptr);  // 用自定义删除器释放资源
    }
};

// SharedPtr 类
template <typename T>
class SharedPtr {
private:
    T* ptr;                  // 指向资源的裸指针
    ControlBlockBase* cb;    // 指向控制块（管理计数和删除器）

    // 释放强引用：计数减1，为0时释放资源
    void release() {
        if (cb) {
            // 强引用计数减1，返回修改前的值
            size_t old_use = cb->use_count.fetch_sub(1);
            if (old_use == 1) {  // 强引用计数即将变为0
                cb->destroy();   // 释放资源
                // 强引用释放后，弱引用计数减1
                size_t old_weak = cb->weak_count.fetch_sub(1);
                if (old_weak == 1) {  // 弱引用计数即将变为0
                    delete cb;        // 释放控制块
                }
            }
            // 重置指针，避免悬垂
            ptr = nullptr;
            cb = nullptr;
        }
    }

public:
    // 1. 构造函数：用裸指针和默认删除器（delete）初始化
    explicit SharedPtr(T* p = nullptr) : ptr(p) {
        if (p) {
            // 默认删除器：delete p
            auto deleter = [](T* x) { delete x; };
            cb = new ControlBlock<T, decltype(deleter)>(p, deleter);
        } else {
            cb = nullptr;
        }
    }

    // 2. 构造函数：支持自定义删除器
    template <typename Deleter>
    SharedPtr(T* p, Deleter d) : ptr(p) {
        if (p) {
            cb = new ControlBlock<T, Deleter>(p, std::move(d));
        } else {
            cb = nullptr;
        }
    }

    // 3. 拷贝构造：共享控制块，强引用计数+1
    SharedPtr(const SharedPtr& other) : ptr(other.ptr), cb(other.cb) {
        if (cb) {
            cb->use_count.fetch_add(1);  // 强引用计数原子+1
        }
    }

    // 4. 移动构造：转移所有权，原对象失效
    SharedPtr(SharedPtr&& other) noexcept : ptr(other.ptr), cb(other.cb) {
        other.ptr = nullptr;
        other.cb = nullptr;
    }

    // 5. 拷贝赋值：释放当前资源，共享新资源
    SharedPtr& operator=(const SharedPtr& other) {
        if (this != &other) {
            release();               // 释放当前资源
            ptr = other.ptr;
            cb = other.cb;
            if (cb) {
                cb->use_count.fetch_add(1);  // 强引用计数+1
            }
        }
        return *this;
    }

    // 6. 移动赋值：释放当前资源，转移新资源所有权
    SharedPtr& operator=(SharedPtr&& other) noexcept {
        if (this != &other) {
            release();               // 释放当前资源
            ptr = other.ptr;
            cb = other.cb;
            other.ptr = nullptr;
            other.cb = nullptr;
        }
        return *this;
    }

    // 7. 析构函数：释放强引用
    ~SharedPtr() {
        release();
    }

    // 8. 重载解引用和成员访问
    T& operator*() const { return *ptr; }
    T* operator->() const { return ptr; }

    // 9. 获取强引用计数
    size_t use_count() const {
        return cb ? cb->use_count.load() : 0;
    }

    // 10. 获取资源指针
    T* get() const { return ptr; }

    // 11. 重置资源（替换为新资源）
    void reset(T* p = nullptr) {
        if (ptr != p) {
            release();  // 释放当前资源
            ptr = p;
            if (p) {
                auto deleter = [](T* x) { delete x; };  // 默认删除器
                cb = new ControlBlock<T, decltype(deleter)>(p, deleter);
            } else {
                cb = nullptr;
            }
        }
    }

    // 允许 WeakPtr 访问私有成员
    friend class WeakPtr<T>;
};

// WeakPtr 类（配合 SharedPtr 使用，不增加强引用计数）
template <typename T>
class WeakPtr {
private:
    T* ptr;                  // 资源指针
    ControlBlockBase* cb;    // 控制块指针

    // 释放弱引用
    void release_weak() {
        if (cb) {
            size_t old_weak = cb->weak_count.fetch_sub(1);
            if (old_weak == 1) {  // 弱引用计数即将变为0
                delete cb;        // 释放控制块
            }
            ptr = nullptr;
            cb = nullptr;
        }
    }

public:
    // 1. 默认构造
    WeakPtr() : ptr(nullptr), cb(nullptr) {}

    // 2. 从 SharedPtr 构造（弱引用计数+1）
    WeakPtr(const SharedPtr<T>& sp) : ptr(sp.ptr), cb(sp.cb) {
        if (cb) {
            cb->weak_count.fetch_add(1);  // 弱引用计数+1
        }
    }

    // 3. 拷贝构造
    WeakPtr(const WeakPtr& other) : ptr(other.ptr), cb(other.cb) {
        if (cb) {
            cb->weak_count.fetch_add(1);  // 弱引用计数+1
        }
    }

    // 4. 拷贝赋值
    WeakPtr& operator=(const WeakPtr& other) {
        if (this != &other) {
            release_weak();          // 释放当前弱引用
            ptr = other.ptr;
            cb = other.cb;
            if (cb) {
                cb->weak_count.fetch_add(1);  // 弱引用计数+1
            }
        }
        return *this;
    }

    // 5. 从 SharedPtr 赋值
    WeakPtr& operator=(const SharedPtr<T>& sp) {
        release_weak();              // 释放当前弱引用
        ptr = sp.ptr;
        cb = sp.cb;
        if (cb) {
            cb->weak_count.fetch_add(1);  // 弱引用计数+1
        }
        return *this;
    }

    // 6. 析构函数：释放弱引用
    ~WeakPtr() {
        release_weak();
    }

    // 7. 检查资源是否有效（强引用计数 > 0）
    bool expired() const {
        return !cb || cb->use_count.load() == 0;
    }

    // 8. 获取 SharedPtr（若资源有效）
    SharedPtr<T> lock() const {
        if (expired()) {
            return SharedPtr<T>(nullptr);  // 资源已释放，返回空
        }
        // 构造 SharedPtr 前先增加强引用计数（避免并发释放）
        cb->use_count.fetch_add(1);
        return SharedPtr<T>(ptr, cb);  // 需要 SharedPtr 支持此构造
    }

private:
    // 供 lock() 调用的私有构造：直接使用已有 ptr 和 cb（强引用计数已+1）
    SharedPtr<T> create_shared() const {
        return SharedPtr<T>(ptr, cb);
    }
};

// 为 SharedPtr 增加从 ptr 和 cb 构造的私有构造函数（供 WeakPtr::lock() 使用）
```

```c++
#include <condition_variable>

using namespace std;

// 共享资源：控制打印顺序的标记（0=线程A打印，1=线程B打印）、打印计数器
int print_flag = 0;  // 0: 线程A（奇数）执行，1: 线程B（偶数）执行
int count = 1;       // 待打印的数字，从1开始
const int MAX_NUM = 10;  // 打印上限

// 互斥锁：保护共享变量print_flag和count
mutex mtx;
// 条件变量：用于线程的阻塞与唤醒
condition_variable cv;

// 线程A：打印奇数
void print_odd() {
    while (count <= MAX_NUM) {
        // 1. 加锁，保护共享变量
        unique_lock<mutex> lock(mtx);

        // 2. 若当前不该自己执行（flag≠0），则阻塞等待，释放锁让其他线程执行
        cv.wait(lock, []() { 
            return print_flag == 0 || count > MAX_NUM;  // 退出条件：该自己执行 或 打印结束
        });

        // 3. 退出条件判断（避免count超限时仍执行）
        if (count > MAX_NUM) {
            break;
        }

        // 4. 执行打印逻辑（奇数）
        cout << "线程A（奇数）: " << count << endl;
        count++;  // 计数器自增

        // 5. 更新标记：下一个该线程B执行
        print_flag = 1;

        // 6. 唤醒所有等待的线程（此时只有线程B会满足条件继续执行）
        cv.notify_all();
    }
}

// 线程B：打印偶数
void print_even() {
    while (count <= MAX_NUM) {
        // 1. 加锁，保护共享变量
        unique_lock<mutex> lock(mtx);

        // 2. 若当前不该自己执行（flag≠1），则阻塞等待
        cv.wait(lock, []() { 
            return print_flag == 1 || count > MAX_NUM;
        });

        // 3. 退出条件判断
        if (count > MAX_NUM) {
            break;
        }

        // 4. 执行打印逻辑（偶数）
        cout << "线程B（偶数）: " << count << endl;
        count++;

        // 5. 更新标记：下一个该线程A执行
        print_flag = 0;

        // 6. 唤醒所有等待的线程
        cv.notify_all();
    }
}

int main() {
    // 创建线程A和线程B
    thread t1(print_odd);
    thread t2(print_even);

    // 等待线程执行完毕
    t1.join();
    t2.join();

    return 0;
}
```
